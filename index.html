def generar_reporte_final(datos_alumnos): print("\n" + "="*40) print("📊 INFORME FINAL DE RESULTADOS") print("="*40) total_aciertos_grupo = 0 total_preguntas_grupo = 0 for alumno in datos_alumnos: nombre = alumno['nombre'] aciertos = alumno['aciertos'] total = alumno['total_preguntas'] # 1. Cálculo de la barra visual porcentaje = (aciertos / total) * 100 longitud_barra = 20 # Caracteres de ancho bloques = int((aciertos / total) * longitud_barra) barra = "█" * bloques + "░" * (longitud_barra - bloques) # Sumamos para el informe global total_aciertos_grupo += aciertos total_preguntas_grupo += total print(f"{nombre.ljust(12)} | {barra} | {aciertos}/{total} ({porcentaje:.1f}%)") # 2. Resumen Global if total_preguntas_grupo > 0: promedio_grupal = (total_aciertos_grupo / total_preguntas_grupo) * 100 print("-" * 40) print(f"RESUMEN DEL GRUPO:") print(f"• Total de alumnos: {len(datos_alumnos)}") print(f"• Promedio de éxito: {promedio_grupal:.2f}%") if promedio_grupal >= 70: print("Estado: ✅ OBJETIVO ALCANZADO") else: print("Estado: ⚠️ REQUIERE REPASO GENERAL") print("="*40) # --- EJEMPLO DE USO --- # Esta lista la irías llenando conforme los alumnos terminan su test resultados = [ {"nombre": "Carlos", "aciertos": 8, "total_preguntas": 10}, {"nombre": "Elena", "aciertos": 10, "total_preguntas": 10}, {"nombre": "Marcos", "aciertos": 5, "total_preguntas": 10}, ] generar_reporte_final(resultados)
